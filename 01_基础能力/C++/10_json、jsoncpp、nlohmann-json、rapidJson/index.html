    10 json、jsoncpp、nlohmann json、rapidJson

    ---

    JSON（JavaScript Object Notation）是一种轻量级的数据交换格式。它易于人类阅读和编写，同时也易于机器解析和生成。JSON 基于一组键值对的数据结构，类似于 Python 中的字典或 C++ 中的 map。

    JSON 基本结构

    **对象（Object）**: 用花括号 {} 包围，并且包含零个或多个键值对，键值对之间用逗号分隔。键是字符串，值可以是任意 JSON 数据类型。

    {
  "name": "Alice",
  "age": 25,
  "isStudent": false
}

    **数组（Array）**: 用方括号 \[\] 包围，并且包含零个或多个值，值之间用逗号分隔。

    [
  "apple",
  "banana",
  "cherry"
]
    • 其他数据类型: 数字、字符串、布尔值、null。
    {
  "number": 123,
  "string": "Hello",
  "boolean": true,
  "nullValue": null
}

    使用 JSONCPP 解析 JSON

    JSONCPP 是一个用 C++ 编写的开源 JSON 解析库。它允许你轻松地解析、生成和操作 JSON 数据。

    安装 JSONCPP

    可以使用包管理工具如 apt 或 brew 安装 JSONCPP，或者从源码编译。

    # For Ubuntu or Debian based systems
sudo apt-get install libjsoncpp-dev

# For macOS
brew install jsoncpp

    示例代码

    解析 JSON 字符串

    #include <iostream>
#include <json/json.h>

int main() {
    std::string rawJson = R"(
    {
        "name": "Alice",
        "age": 25,
        "isStudent": false,
        "scores": [85, 92, 78]
    })";

    Json::CharReaderBuilder readerBuilder;
    Json::CharReader* reader = readerBuilder.newCharReader();
    Json::Value root;
    std::string errs;

    if (reader->parse(rawJson.c_str(), rawJson.c_str() + rawJson.size(), &root, &errs)) {
        std::cout << "Name: " << root["name"].asString() << std::endl;
        std::cout << "Age: " << root["age"].asInt() << std::endl;
        std::cout << "Is student: " << root["isStudent"].asBool() << std::endl;

        const Json::Value scores = root["scores"];
        std::cout << "Scores: ";
        for (const auto& score : scores) {
            std::cout << score.asInt() << " ";
        }
        std::cout << std::endl;
    } else {
        std::cerr << "Failed to parse JSON: " << errs << std::endl;
    }

    delete reader;
    return 0;
}

    生成 JSON 字符串

    #include <iostream>
#include <json/json.h>

int main() {
    Json::Value root;
    root["name"] = "Alice";
    root["age"] = 25;
    root["isStudent"] = false;

    Json::Value scores(Json::arrayValue);
    scores.append(85);
    scores.append(92);
    scores.append(78);
    root["scores"] = scores;

    Json::StreamWriterBuilder writerBuilder;
    std::string outputJson = Json::writeString(writerBuilder, root);
    std::cout << outputJson << std::endl;

    return 0;
}

    主要类和方法

    • Json::Value: 代表一个 JSON 值，可以是对象、数组、字符串、数字、布尔值或 null。
    • Json::CharReaderBuilder: 用于创建 JSON 解析器。
    • Json::StreamWriterBuilder: 用于创建 JSON 序列化器。
    • asString(), asInt(), asBool(): 将 JSON 值转换为相应的 C++ 类型。
    • append(): 向 JSON 数组添加元素。
    • operator[]: 访问或修改 JSON 对象的键值对或数组元素。

    复杂的 JSON 示例

    假设我们有如下的 JSON 数据：

    {
    "company": {
        "name": "TechCorp",
        "employees": [
            {
                "name": "Alice",
                "age": 30,
                "skills": ["C++", "Python", "SQL"]
            },
            {
                "name": "Bob",
                "age": 25,
                "skills": ["Java", "JavaScript"]
            }
        ],
        "departments": {
            "R&D": {
                "budget": 1000000,
                "projects": ["AI", "Quantum Computing"]
            },
            "HR": {
                "budget": 500000,
                "projects": ["Recruitment", "Training"]
            }
        }
    }
}

    解析这个 JSON 数据

    #include <iostream>
#include <json/json.h>

int main() {
    std::string rawJson = R"(
    {
        "company": {
            "name": "TechCorp",
            "employees": [
                {
                    "name": "Alice",
                    "age": 30,
                    "skills": ["C++", "Python", "SQL"]
                },
                {
                    "name": "Bob",
                    "age": 25,
                    "skills": ["Java", "JavaScript"]
                }
            ],
            "departments": {
                "R&D": {
                    "budget": 1000000,
                    "projects": ["AI", "Quantum Computing"]
                },
                "HR": {
                    "budget": 500000,
                    "projects": ["Recruitment", "Training"]
                }
            }
        }
    })";

    Json::CharReaderBuilder readerBuilder;
    Json::CharReader* reader = readerBuilder.newCharReader();
    Json::Value root;
    std::string errs;

    if (reader->parse(rawJson.c_str(), rawJson.c_str() + rawJson.size(), &root, &errs)) {
        const Json::Value company = root["company"];
        std::cout << "Company Name: " << company["name"].asString() << std::endl;

        const Json::Value employees = company["employees"];
        for (const auto& employee : employees) {
            std::cout << "Employee Name: " << employee["name"].asString() << ", Age: " << employee["age"].asInt() << std::endl;
            std::cout << "Skills: ";
            for (const auto& skill : employee["skills"]) {
                std::cout << skill.asString() << " ";
            }
            std::cout << std::endl;
        }

        const Json::Value departments = company["departments"];
        for (const auto& department : departments.getMemberNames()) {
            std::cout << "Department: " << department << std::endl;
            std::cout << "Budget: " << departments[department]["budget"].asInt() << std::endl;
            std::cout << "Projects: ";
            for (const auto& project : departments[department]["projects"]) {
                std::cout << project.asString() << " ";
            }
            std::cout << std::endl;
        }
    } else {
        std::cerr << "Failed to parse JSON: " << errs << std::endl;
    }

    delete reader;
    return 0;
}

    创建一个类似结构的 JSON 数据

    #include <iostream>
#include <json/json.h>

int main() {
    Json::Value root;
    Json::Value company;
    company["name"] = "TechCorp";

    Json::Value employees(Json::arrayValue);

    Json::Value employee1;
    employee1["name"] = "Alice";
    employee1["age"] = 30;
    Json::Value skills1(Json::arrayValue);
    skills1.append("C++");
    skills1.append("Python");
    skills1.append("SQL");
    employee1["skills"] = skills1;

    Json::Value employee2;
    employee2["name"] = "Bob";
    employee2["age"] = 25;
    Json::Value skills2(Json::arrayValue);
    skills2.append("Java");
    skills2.append("JavaScript");
    employee2["skills"] = skills2;

    employees.append(employee1);
    employees.append(employee2);

    company["employees"] = employees;

    Json::Value departments;
    Json::Value rd;
    rd["budget"] = 1000000;
    Json::Value projects1(Json::arrayValue);
    projects1.append("AI");
    projects1.append("Quantum Computing");
    rd["projects"] = projects1;

    Json::Value hr;
    hr["budget"] = 500000;
    Json::Value projects2(Json::arrayValue);
    projects2.append("Recruitment");
    projects2.append("Training");
    hr["projects"] = projects2;

    departments["R&D"] = rd;
    departments["HR"] = hr;

    company["departments"] = departments;

    root["company"] = company;

    Json::StreamWriterBuilder writerBuilder;
    std::string outputJson = Json::writeString(writerBuilder, root);
    std::cout << outputJson << std::endl;

    return 0;
}

    主要步骤

    1. 解析 JSON 数据
      • 使用 Json::CharReaderBuilder 创建解析器。
      • 使用 reader->parse 解析 JSON 字符串到 Json::Value 对象中。
      • 使用 asString(), asInt() 等方法提取数据。
    2. 创建 JSON 数据
      • 创建 Json::Value 对象，并使用 operator[] 和 append() 构建 JSON 结构。
      • 使用 Json::StreamWriterBuilder 和 Json::writeString 将 JSON 对象序列化为字符串。

    使用 nlohmann-json 解析 JSON nlohmann-json 是一个用于处理 JSON 的 C++ 库，它以其简洁和易用性而闻名。以下是详细讲解如何使用 nlohmann-json 进行 JSON 数据的解析和生成的步骤。

    安装 nlohmann-json

    你可以通过包管理工具安装 nlohmann-json，或者直接将头文件包含到你的项目中。

    使用包管理工具安装

    # For Ubuntu or Debian based systems
sudo apt-get install nlohmann-json3-dev

# For macOS
brew install nlohmann-json

    从源码安装

    你可以从 GitHub 下载 nlohmann-json 的头文件，并将其包含到你的项目中：

    #include <nlohmann/json.hpp>

    解析 JSON 数据

    以下是如何使用 nlohmann-json 解析 JSON 数据的示例：

    #include <iostream>
#include <nlohmann/json.hpp>

int main() {
    // JSON 字符串
    std::string jsonString = R"(
    {
        "company": {
            "name": "TechCorp",
            "employees": [
                {
                    "name": "Alice",
                    "age": 30,
                    "skills": ["C++", "Python", "SQL"]
                },
                {
                    "name": "Bob",
                    "age": 25,
                    "skills": ["Java", "JavaScript"]
                }
            ],
            "departments": {
                "R&D": {
                    "budget": 1000000,
                    "projects": ["AI", "Quantum Computing"]
                },
                "HR": {
                    "budget": 500000,
                    "projects": ["Recruitment", "Training"]
                }
            }
        }
    })";

    // 解析 JSON 字符串
    nlohmann::json jsonData = nlohmann::json::parse(jsonString);

    // 访问 JSON 数据
    std::cout << "Company Name: " << jsonData["company"]["name"] << std::endl;

    for (const auto& employee : jsonData["company"]["employees"]) {
        std::cout << "Employee Name: " << employee["name"] << ", Age: " << employee["age"] << std::endl;
        std::cout << "Skills: ";
        for (const auto& skill : employee["skills"]) {
            std::cout << skill << " ";
        }
        std::cout << std::endl;
    }

    for (const auto& department : jsonData["company"]["departments"].items()) {
        std::cout << "Department: " << department.key() << std::endl;
        std::cout << "Budget: " << department.value()["budget"] << std::endl;
        std::cout << "Projects: ";
        for (const auto& project : department.value()["projects"]) {
            std::cout << project << " ";
        }
        std::cout << std::endl;
    }

    return 0;
}

    创建 JSON 数据

    以下是如何使用 nlohmann-json 创建 JSON 数据的示例：

    #include <iostream>
#include <nlohmann/json.hpp>

int main() {
    // 创建 JSON 对象
    nlohmann::json jsonData;

    jsonData["company"]["name"] = "TechCorp";

    jsonData["company"]["employees"] = {
        {{"name", "Alice"}, {"age", 30}, {"skills", {"C++", "Python", "SQL"}}},
        {{"name", "Bob"}, {"age", 25}, {"skills", {"Java", "JavaScript"}}}
    };

    jsonData["company"]["departments"] = {
        {"R&D", {{"budget", 1000000}, {"projects", {"AI", "Quantum Computing"}}}},
        {"HR", {{"budget", 500000}, {"projects", {"Recruitment", "Training"}}}}
    };

    // 输出 JSON 字符串
    std::cout << jsonData.dump(4) << std::endl;

    return 0;
}

    主要步骤

    解析 JSON 数据

    1. 包含头文件：#include
    2. 解析 JSON 字符串：nlohmann::json jsonData = nlohmann::json::parse(jsonString);
    3. 访问数据：使用类似数组的语法访问 JSON 数据，例如 jsonData["key"].

    创建 JSON 数据

    1. 创建 JSON 对象：nlohmann::json jsonData;
    2. 添加数据：使用类似数组的语法添加数据，例如 jsonData["key"] = value;
    3. 输出 JSON 字符串：使用 dump() 方法将 JSON 对象转换为字符串，例如 jsonData.dump(4);，其中 4 表示缩进的空格数。

    主要类和方法

    • nlohmann::json：这是 JSON 对象的核心类。
    • parse()：用于解析 JSON 字符串。
    • dump()：用于将 JSON 对象序列化为字符串。
    • operator[]：用于访问和修改 JSON 对象的键值对。

    追加 你可以使用 nlohmann::json 库来打开一个 JSON 文件，并将新的数据追加到现有的数组中。以下是一个示例，展示了如何实现这一点：

    示例代码

    假设你有一个 JSON 文件 data.json，并且你想将新的数组元素追加到这个文件中的现有数组后面。

    #include <iostream>
#include <fstream>
#include <nlohmann/json.hpp>

int main() {
    // 定义文件路径
    std::string filePath = "data.json";

    // 打开 JSON 文件
    std::ifstream inputFile(filePath);
    if (!inputFile.is_open()) {
        std::cerr << "无法打开文件: " << filePath << std::endl;
        return 1;
    }

    // 解析现有的 JSON 数据
    nlohmann::json jsonData;
    try {
        inputFile >> jsonData;
    } catch (const nlohmann::json::parse_error& e) {
        std::cerr << "解析 JSON 文件失败: " << e.what() << std::endl;
        return 1;
    }
    inputFile.close(); // 关闭输入文件

    // 确保 jsonData 是一个数组
    if (!jsonData.is_array()) {
        std::cerr << "JSON 文件格式错误：根元素不是数组。" << std::endl;
        return 1;
    }

    // 定义要追加的新的 JSON 对象
    nlohmann::json newEntry = {
        {"timestamp", 2398700},
        {"framenumber", 20},
        {"ori_rect", {
            {"left", 780.0},
            {"top", 410.0},
            {"width", 250.0},
            {"height", 670.0}
        }},
        {"new_rect", {
            {"left", 350.0},
            {"top", 0.0},
            {"width", 1080.0},
            {"height", 1920.0}
        }},
        {"filename", "/home/zhengwei/work/projects/miniproject/deploy/images/Person-0-45/0_20_250x670_2398700.jpg"}
    };

    // 追加新的 JSON 对象到数组中
    jsonData.push_back(newEntry);

    // 将更新后的 JSON 数据写回文件
    std::ofstream outputFile(filePath);
    if (!outputFile.is_open()) {
        std::cerr << "无法打开文件进行写入: " << filePath << std::endl;
        return 1;
    }
    outputFile << jsonData.dump(4); // 使用缩进为4的格式写回文件
    outputFile.close(); // 关闭输出文件

    std::cout << "JSON 数据已成功追加并写入文件。" << std::endl;

    return 0;
}

    代码说明

    1. 打开和读取 JSON 文件：
      • 使用 std::ifstream 打开现有的 JSON 文件。
      • 将文件内容解析为 nlohmann::json 对象。
    2. 检查文件内容是否为数组：
      • 确保 JSON 文件的根元素是一个数组，因为我们要向这个数组追加元素。
    3. 创建新的 JSON 对象：
      • 新的数据被定义为一个 nlohmann::json 对象，并按要求设置各个字段。
    4. 追加新的 JSON 对象：
      • 使用 push_back() 方法将新的 JSON 对象追加到数组中。
    5. 写回更新后的 JSON 数据：
      • 使用 std::ofstream 打开文件，并将更新后的 JSON 数据写回文件。
      • 使用 dump(4) 将 JSON 数据格式化为带有缩进的字符串，并写入文件中。
    6. 关闭文件：
      • 在读写操作完成后关闭文件。

    注意事项

    • 异常处理：在解析 JSON 文件时处理可能出现的异常，避免程序崩溃。
    • 文件路径：确保文件路径正确，并且程序具有读写权限。

    这种方法可以帮助你在处理 JSON 文件时动态追加数据，并确保数据结构的完整性。

    RapidJSON 是一个高性能的 JSON 解析和生成库。它是一个纯 C++ 实现的库，具有很高的解析速度和较小的内存占用。以下是使用 RapidJSON 的基本示例，包括解析 JSON 数据和创建 JSON 数据。

    1. 解析 JSON 数据

    首先，确保你已经安装并正确配置了 RapidJSON 库。 以下是解析 JSON 数据的示例： cpp

    #include "rapidjson/document.h" // 包含 RapidJSON 的头文件
#include <iostream>

int main() {
    // 假设有一个 JSON 字符串
    const char* json = R"({
        "name": "Alice",
        "age": 25,
        "is_student": false,
        "skills": ["C++", "Python", "Machine Learning"]
    })";

    // 创建一个 RapidJSON Document 对象
    rapidjson::Document document;

    // 解析 JSON 字符串
    if (document.Parse(json).HasParseError()) {
        std::cerr << "JSON 解析错误!" << std::endl;
        return 1;
    }

    // 读取 JSON 数据
    if (document.HasMember("name") && document["name"].IsString()) {
        std::cout << "Name: " << document["name"].GetString() << std::endl;
    }

    if (document.HasMember("age") && document["age"].IsInt()) {
        std::cout << "Age: " << document["age"].GetInt() << std::endl;
    }

    if (document.HasMember("is_student") && document["is_student"].IsBool()) {
        std::cout << "Is student: " << (document["is_student"].GetBool() ? "Yes" : "No") << std::endl;
    }

    if (document.HasMember("skills") && document["skills"].IsArray()) {
        std::cout << "Skills: ";
        const rapidjson::Value& skills = document["skills"];
        for (rapidjson::SizeType i = 0; i < skills.Size(); i++) {
            std::cout << skills[i].GetString() << (i < skills.Size() - 1 ? ", " : "");
        }
        std::cout << std::endl;
    }

    return 0;
}

    2. 创建 JSON 数据

    以下是使用 RapidJSON 创建 JSON 数据的示例： cpp

    #include "rapidjson/document.h"
#include "rapidjson/writer.h"
#include "rapidjson/stringbuffer.h"
#include <iostream>

int main() {
    // 创建一个 RapidJSON Document 对象
    rapidjson::Document document;
    document.SetObject(); // 设置为对象类型

    // 创建一个 JSON 分配器（用于分配内存）
    rapidjson::Document::AllocatorType& allocator = document.GetAllocator();

    // 添加数据到 JSON 对象
    document.AddMember("name", "Bob", allocator);
    document.AddMember("age", 30, allocator);
    document.AddMember("is_student", false, allocator);

    // 创建一个 JSON 数组并添加元素
    rapidjson::Value skills(rapidjson::kArrayType);
    skills.PushBack("Java", allocator).PushBack("Go", allocator).PushBack("JavaScript", allocator);
    document.AddMember("skills", skills, allocator);

    // 转换 JSON 对象为字符串
    rapidjson::StringBuffer buffer;
    rapidjson::Writer<rapidjson::StringBuffer> writer(buffer);
    document.Accept(writer);

    // 输出 JSON 字符串
    std::cout << "Created JSON: " << buffer.GetString() << std::endl;

    return 0;
}

    解释

    1. 解析 JSON 数据:
      • 使用 rapidjson::Document 解析 JSON 字符串。
      • 使用 HasMember 方法检查 JSON 中是否存在某个字段。
      • 使用 GetString()、GetInt() 等方法获取字段值。
    2. 创建 JSON 数据:
      • 使用 rapidjson::Document 创建一个新的 JSON 对象。
      • 使用 AddMember 方法向 JSON 对象添加字段和值。
      • 使用 PushBack 方法向 JSON 数组添加元素。
      • 使用 rapidjson::Writer 将 JSON 对象转换为字符串。
